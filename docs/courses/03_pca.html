Chemometrics Part 3 — Principal Component Analysis

Latent structure, score plots, and loadings of grape berry spectra

Learning objectives

After completing this module you should be able to:

  1. Justify the choice of centring without unit-variance scaling for NIR spectra.
  2. Interpret a scree plot and choose an appropriate number of PCs.
  3. Read a score scatter plot and link cluster structure to chemical variation.
  4. Interpret loading vectors as spectral fingerprints of latent components.
  5. Quantify the predictive relevance of each PC via correlation with the response.

Dataset citation: Ryckewaert, M. et al. (2023). Dataset containing spectral data from hyperspectral imaging and sugar content measurements of grape berries in various maturity stages. Data in Brief. https://doi.org/10.1016/j.dib.2022.108822

Introduction

Principal Component Analysis (PCA) is the foundational dimensionality-reduction method in chemometrics. A spectral matrix \(\mathbf{X} \in \mathbb{R}^{n \times p}\) (274 samples, 204 wavelengths) lives in a 204-dimensional space, but the true chemical variation is far lower-dimensional. PCA finds orthogonal directions of maximum variance:

\[\mathbf{X} = \mathbf{T}\mathbf{P}^\top + \mathbf{E}\]

  • \(\mathbf{T} \in \mathbb{R}^{n \times k}\) — scores: sample coordinates in the PC subspace
  • \(\mathbf{P} \in \mathbb{R}^{p \times k}\) — loadings: spectral directions (unit vectors)
  • \(\mathbf{E}\) — residual matrix (unexplained variation)
import pandas as pd
import numpy as np
from sklearn.decomposition import PCA
import plotly.graph_objects as go
from plotly.subplots import make_subplots

NAVY  = '#1b3a5c'
TEAL  = '#2a7f7f'
GREEN = '#78BE20'
GRAY  = '#868e96'
COLORS_VAR  = {'SYRAH': NAVY, 'MAUZAC': TEAL, 'FER': GREEN}
MARKERS_VAR = {'SYRAH': 'triangle-up', 'MAUZAC': 'square', 'FER': 'circle'}
df = pd.read_csv('data/chemometrics/DATASET.csv', sep=';', index_col=0)

variety = df['Variety']
y = df['Sugar content (g/l)']

spectral_cols = [c for c in df.columns if c.startswith('x.')]
X = df[spectral_cols].values
wavelengths = np.array([float(c.replace('x.', '', 1)) for c in spectral_cols])

Preprocessing: centring without unit-variance scaling

Before PCA, we centre the data (subtract the column mean):

\[\tilde{x}_{ik} = x_{ik} - \bar{x}_k, \quad \bar{x}_k = \frac{1}{n}\sum_{i=1}^n x_{ik}\]

We omit unit-variance scaling deliberately. In NIR spectroscopy, spectral channels do not contribute equally to variance — regions near water absorption bands (~970 nm) or the VIS–NIR transition (~700 nm) naturally show higher variability reflecting real chemical signal. Scaling to unit variance would up-weight noise-dominated channels and destroy the physically meaningful variance structure.

X_centred = X - X.mean(axis=0)

n_components = 20
pca = PCA(n_components=n_components)
T   = pca.fit_transform(X_centred)   # scores:   (274, 20)
P   = pca.components_                # loadings:  (20, 204)
evr = pca.explained_variance_ratio_

print("Cumulative explained variance:")
for k in [1, 2, 3, 5, 10]:
    print(f"  First {k:2d} PCs: {evr[:k].sum()*100:.1f}%")
Cumulative explained variance:
  First  1 PCs: 79.5%
  First  2 PCs: 96.2%
  First  3 PCs: 99.0%
  First  5 PCs: 99.8%
  First 10 PCs: 100.0%

Scree plot

The scree plot shows variance per PC (bars) and the cumulative sum (line). An elbow in the curve suggests the intrinsic dimensionality of the dataset. The 95% threshold indicates the minimum number of PCs to retain most spectral variance.

fig = make_subplots(specs=[[{'secondary_y': True}]])

fig.add_trace(
    go.Bar(x=list(range(1, n_components + 1)), y=evr * 100,
           name='Individual PC', marker_color=NAVY, opacity=0.75),
    secondary_y=False
)
fig.add_trace(
    go.Scatter(x=list(range(1, n_components + 1)), y=evr.cumsum() * 100,
               mode='lines+markers', name='Cumulative',
               line=dict(color=TEAL, width=2), marker=dict(size=5, color=TEAL)),
    secondary_y=True
)
fig.add_hline(y=95, line_dash='dash', line_color=GREEN,
              annotation_text='95%', annotation_position='right',
              secondary_y=True)

fig.update_xaxes(title_text='Principal Component', dtick=1)
fig.update_yaxes(title_text='Explained variance (%)', secondary_y=False, color=NAVY)
fig.update_yaxes(title_text='Cumulative explained variance (%)',
                 secondary_y=True, color=TEAL, range=[0, 102])
fig.update_layout(title='PCA scree plot — centred grape berry spectra',
                  template='simple_white', height=400,
                  legend=dict(orientation='h', y=1.1))
fig.show()

Score scatter plot: PC1 vs PC2

The score plot projects each sample onto the first two PCs. Colour encodes sugar content; marker shape encodes variety — revealing whether the PCs align with the sugar gradient and whether varieties cluster separately.

fig = go.Figure()
for var in variety.unique():
    mask = (variety == var).values
    fig.add_trace(go.Scatter(
        x=T[mask, 0], y=T[mask, 1], mode='markers',
        marker=dict(
            symbol=MARKERS_VAR[var], size=8,
            color=y.values[mask], colorscale='RdYlBu_r',
            cmin=float(y.min()), cmax=float(y.max()),
            colorbar=dict(title='Sugar content (g/L)', thickness=15, len=0.9)
                if var == list(variety.unique())[0] else dict(),
            showscale=(var == list(variety.unique())[0]),
            line=dict(color='white', width=0.5)
        ),
        name=var,
        hovertemplate='%{text}<extra>' + var + '</extra>',
        text=[f'Sugar: {s:.1f} g/L' for s in y.values[mask]]
    ))
fig.add_hline(y=0, line_dash='dash', line_color=GRAY, line_width=0.6)
fig.add_vline(x=0, line_dash='dash', line_color=GRAY, line_width=0.6)
fig.update_layout(
    xaxis_title=f'PC1  ({evr[0]*100:.1f}% var. expl.)',
    yaxis_title=f'PC2  ({evr[1]*100:.1f}% var. expl.)',
    title='PCA score plot — coloured by sugar content',
    template='simple_white', height=480,
    legend=dict(title='Variety')
)
fig.show()

Loadings

Loading vectors \(\mathbf{p}_k\) indicate the contribution of each wavelength to PC \(k\). Peaks mark spectral regions driving that component, linkable to known molecular absorption bands (~680 nm chlorophyll red edge, ~970 nm water O–H overtone).

colors_pc = [NAVY, TEAL, GREEN]
fig = make_subplots(rows=3, cols=1, shared_xaxes=True,
                    subplot_titles=[f'PC{i+1} ({evr[i]*100:.1f}% explained)' for i in range(3)],
                    vertical_spacing=0.08)
for i, color in enumerate(colors_pc):
    loading = P[i]
    r, g, b = int(color[1:3], 16), int(color[3:5], 16), int(color[5:7], 16)
    fig.add_trace(
        go.Scatter(x=wavelengths, y=loading, mode='lines', fill='tozeroy',
                   fillcolor=f'rgba({r},{g},{b},0.15)',
                   line=dict(color=color, width=1.3), showlegend=False),
        row=i + 1, col=1
    )
    fig.add_hline(y=0, line_dash='dash', line_color=GRAY, line_width=0.6,
                  row=i + 1, col=1)
    fig.update_yaxes(title_text=f'PC{i+1} loading', row=i + 1, col=1)
fig.update_xaxes(title_text='Wavelength (nm)', row=3, col=1)
fig.update_layout(title='First three PCA loading vectors',
                  template='simple_white', height=540)
fig.show()

Correlation of PC scores with sugar content

correlations = np.array([
    np.corrcoef(T[:, k], y.values)[0, 1]
    for k in range(n_components)
])
bar_colors = [TEAL if c > 0 else NAVY for c in correlations]

fig = go.Figure(go.Bar(
    x=list(range(1, n_components + 1)), y=correlations,
    marker_color=bar_colors, opacity=0.8,
    hovertemplate='PC%{x}: r = %{y:.3f}<extra></extra>'
))
fig.add_hline(y=0, line_color='black', line_width=0.8)
fig.update_layout(
    xaxis_title='Principal Component', xaxis=dict(dtick=1),
    yaxis_title='Pearson r with sugar content',
    title='Correlation of PC scores with sugar content (g/L)',
    template='simple_white', height=350
)
fig.show()

top3 = np.argsort(np.abs(correlations))[::-1][:3]
print("Top 3 PCs most correlated with sugar content:")
for k in top3:
    print(f"  PC{k+1}: r = {correlations[k]:.3f}  ({evr[k]*100:.1f}% var. expl.)")
Top 3 PCs most correlated with sugar content:
